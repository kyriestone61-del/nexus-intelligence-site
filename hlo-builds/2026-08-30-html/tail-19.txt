    const aw=(a.concept_tags||[]).some(t=>weak.includes(t))?0:1,bw=(b.concept_tags||[]).some(t=>weak.includes(t))?0:1;
    if(aw!==bw)return aw-bw;return Math.abs(difficultyRank(a.difficulty)-rank)-Math.abs(difficultyRank(b.difficulty)-rank)
  });
