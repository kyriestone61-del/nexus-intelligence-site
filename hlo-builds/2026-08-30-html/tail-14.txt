function go(v){
  if(window.HLOCloud?.isOwner===false && ['nexus','statecraft','career','trading','tradelog','plan'].includes(v)) v='dashboard';
  if(!v || !document.getElementById(v)) return;
  document.querySelectorAll('.view').forEach(x=>x.classList.remove('active'));
  document.querySelectorAll('nav button[data-view]').forEach(x=>x.classList.toggle('active',x.dataset.view===v));
  document.getElementById(v).classList.add('active');
  document.querySelectorAll('.nav-group').forEach(g=>{
    const has=!!g.querySelector(`button[data-view="${v}"]`);
    g.classList.toggle('has-active',has);
    if(has) g.classList.add('open');
  });
  window.scrollTo({top:0,behavior:'smooth'});
}

function sourceLinks(ids){
  return ids.map(id=>{
    const s=SOURCES.find(x=>x.id===id); if(!s)return '';
    return `<a class="badge" href="${s.url}" target="_blank" rel="noopener">${s.title}</a>`;
  }).join(' ');
}

function academyModuleProgress(m){
  const done=m.lessonIds.filter(id=>state.done[id]).length;
  const lessonPct=m.lessonIds.length?Math.round(done/m.lessonIds.length*100):0;
  const quiz=Number(academyQuizScores[m.id]||0);
  return {done,total:m.lessonIds.length,lessonPct,quiz,mastered:lessonPct===100&&quiz>=80};
}
function academyOverall(){
  const ps=ACADEMY_MODULES.map(academyModuleProgress);
  const avg=Math.round(ps.reduce((s,p)=>s+(p.lessonPct*.7+p.quiz*.3),0)/ps.length);
  const mastered=ps.filter(p=>p.mastered).length;
  return {avg,mastered};
}

function renderAcademy(){
  const list=document.getElementById('academyModuleList'), content=document.getElementById('academyModuleContent');
  if(!list||!content)return;
  const mods=ACADEMY_MODULES.filter(m=>!academyStageFilter||m.level===academyStageFilter);
  if(academyStageFilter && !mods.some(m=>m.id===academyActiveModule)) academyActiveModule=mods[0]?.id||'A1';
  list.innerHTML=mods.map(m=>{
    const p=academyModuleProgress(m);
    return `<button class="academy-module-btn ${m.id===academyActiveModule?'active':''}" onclick="openAcademyModule('${m.id}')">
      <span class="module-num">${m.id} · ${m.level}</span>
      <span class="module-title">${m.title}</span>
      <span class="module-meta"><span>${m.duration}</span><span>${p.mastered?'Mastered':p.lessonPct+'% lessons · '+p.quiz+'% quiz'}</span></span>
    </button>`;
  }).join('');
  const o=academyOverall();
  const pct=document.getElementById('academyMasteryPct'),bar=document.getElementById('academyMasteryBar'),txt=document.getElementById('academyMasteryText');
  if(pct)pct.textContent=o.avg+'%';if(bar)bar.style.width=o.avg+'%';if(txt)txt.textContent=`${o.mastered} of ${ACADEMY_MODULES.length} modules mastered`;
  renderAcademyModule(academyActiveModule);
}
