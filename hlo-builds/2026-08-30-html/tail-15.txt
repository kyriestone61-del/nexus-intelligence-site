function renderModuleQuizStart(moduleId){
  const host=document.getElementById('moduleQuizHost'),m=ACADEMY_MODULES.find(x=>x.id===moduleId);if(!host||!m)return;
  const prev=Number(academyQuizScores[moduleId]||0);
  host.innerHTML=`<div class="eyebrow">Module knowledge check</div><h2>${m.quiz.length}-question quiz</h2><p class="muted">Target: 80%+. Explanations appear after every answer. Your best score is retained.</p>${prev?`<div class="pill">Best score: ${prev}%</div>`:''}<br><button class="primary" onclick="startModuleQuiz('${moduleId}')">${prev?'Retake module quiz':'Start module quiz'}</button>`;
}
function startModuleQuiz(moduleId){
  const m=ACADEMY_MODULES.find(x=>x.id===moduleId);if(!m)return;
  academyQuizState={moduleId,idx:0,score:0,answers:[]};renderModuleQuizQuestion();
}
window.startModuleQuiz=startModuleQuiz;
function renderModuleQuizQuestion(){
  const s=academyQuizState,host=document.getElementById('moduleQuizHost');if(!s||!host)return;
  const m=ACADEMY_MODULES.find(x=>x.id===s.moduleId);if(s.idx>=m.quiz.length)return finishModuleQuiz();
  const q=m.quiz[s.idx];
  host.innerHTML=`<div class="small muted">${m.id} · Question ${s.idx+1} of ${m.quiz.length}</div><h3>${q.q}</h3>${q.choices.map((c,i)=>`<button class="module-quiz-choice" onclick="answerModuleQuiz(${i})">${c}</button>`).join('')}<div id="moduleQuizExplain"></div>`;
}
function answerModuleQuiz(i){
  const s=academyQuizState,m=ACADEMY_MODULES.find(x=>x.id===s.moduleId),q=m.quiz[s.idx],correct=i===q.a;
  if(correct)s.score++;s.answers.push({question:s.idx,selected:i,correct});
  [...document.querySelectorAll('.module-quiz-choice')].forEach((b,j)=>{b.disabled=true;if(j===q.a)b.classList.add('correct');else if(j===i)b.classList.add('wrong')});
  const e=document.getElementById('moduleQuizExplain');e.innerHTML=`<div class="callout"><b>${correct?'Correct':'Not quite'}.</b> ${q.ex}</div><button class="primary" onclick="nextModuleQuizQuestion()">Continue</button>`;
}
window.answerModuleQuiz=answerModuleQuiz;
function nextModuleQuizQuestion(){academyQuizState.idx++;renderModuleQuizQuestion()}
window.nextModuleQuizQuestion=nextModuleQuizQuestion;
function finishModuleQuiz(){
  const s=academyQuizState,m=ACADEMY_MODULES.find(x=>x.id===s.moduleId),host=document.getElementById('moduleQuizHost');
  const pct=Math.round(s.score/m.quiz.length*100),best=Math.max(Number(academyQuizScores[m.id]||0),pct);
  academyQuizScores[m.id]=best;localStorage.setItem('hlo_module_quiz_scores',JSON.stringify(academyQuizScores));
  host.innerHTML=`<div class="eyebrow">Module result</div><div class="score">${pct}%</div><p>${pct>=80?'Knowledge gate passed. Now make sure the applied exercise and related units are complete.':'Review the deep lessons and retake. Treat mistakes as a map of what to study next.'}</p><button class="primary" onclick="startModuleQuiz('${m.id}')">Retake</button>`;
  if(window.HLOCloud?.recordModuleQuiz)window.HLOCloud.recordModuleQuiz(m.id,pct,s.answers,m.level);
  renderAcademy();
}
