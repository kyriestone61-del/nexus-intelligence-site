function ascension(){
  const p=completionPct(), q=cloud.quizzes.length?Math.max(...cloud.quizzes.map(x=>Number(x.score)||0)):C.state.quizBest||0;
  if(p>=90&&q>=85)return 'Allocator';
  if(p>=70&&q>=85)return 'Owner';
  if(p>=45&&q>=80)return 'Architect';
  if(p>=20&&q>=75)return 'Operator';
  return 'Beginner';
}
